  ylabel={entropy\_rate\_norm},
  symbolic x coords={the,and,i,to,of,is,it,that,love,king,heart,sweet,man,osric,banquo},

  % 规则阈值 0.93
  \addplot[dashed, line width=0.8pt, color=black!65, mark=none]

  \addlegendentry{0.93 threshold}

\caption{Rate entropy as a stop-word test: function words cluster near 1 (all
above 0.95), proper nouns with df${}=1$ are zero by definition, and content
words fall in the lower band 0.83--0.94. A single 0.93 line separates function
words from the rest; the few content words above the line are still blocked by
the df-ratio gate ($\ge 0.4$), so the feature can replace manual review.}%
\label{fig:entropy}
