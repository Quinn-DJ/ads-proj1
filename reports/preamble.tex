% ---------- Fonts and language ----------

% ---------- Report macros ----------
% Highlighted finding: \finding{one-sentence conclusion}
\newcommand{\finding}[1]{%
  \par\medskip\noindent
  \begin{minipage}{\linewidth}%
    \colorbox{codebg}{\parbox{\dimexpr\linewidth-2\fboxsep}{%
      \raggedright\noindent\textbf{\color{mseblue}Finding: }#1}}%
  \end{minipage}%
  \par\medskip
}
